import java.util.*;
public class Main{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        
        int t = sc.nextInt();
        
        while(t > 0){
            long n = sc.nextLong();
            boolean found = false;
            for(int i=0; 12*i<=n; i++){
                long a = n - 12*i;
                if(isPalindrome(a)){
                    System.out.println(a+" "+12*i);
                    found = true;
                    break;
                }
            }
            if(!found){
                System.out.println("-1");
            }
            t--;
        }
    }

    public static boolean isPalindrome(long n){
        String s = Long.toString(n);
        int left = 0;
        int right = s.length()-1;
        while(left < right){
            if(s.charAt(left) != s.charAt(right)){
                return false;
            }
            left++;
            right--;
        }
        return true;
    }
}
