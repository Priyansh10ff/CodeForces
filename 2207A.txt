import java.util.*;
public class Main{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        
        int t = sc.nextInt();
        
        while(t > 0){
            int n = sc.nextInt();
            String s = sc.next();
            
            int ones = 0;
            int minOne = 0;
            int maxone = 0;

            for(int i=0; i<n; i++){
                if(s.charAt(i) == '1'){
                    ones++;
                }
            }
            maxone = ones;

            for(int i=1; i<n-1; i++){
                if(s.charAt(i) == '0' && s.charAt(i+1) == '1' && s.charAt(i-1) == '1'){
                    maxone++;
                }
            }

            int i = 0;

            while(i < n){
                if(s.charAt(i) == '1'){
                    int j = i;
                    while(j < n && s.charAt(j) == '1'){
                        j++;
                    }
                    int len = j-i;
                    minOne += Math.min(len, 2);

                    i = j;
                }
                else{
                    i++;
                }
            }
            System.out.println(minOne + " " + maxone);
            t--;
        }
    }
}
